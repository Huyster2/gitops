version: "3.5"

services:
  mongo:
    container_name: user_management_dev
    image: mongo:4.4.2
      #    ports:
      #      - 27017:27017
    command: mongod --port 27017

  mongo_seed:
    image: 'pkosiec/mongo-seeding:3.0.0'
    environment:
      - DROP_DATABASE=true
      - REPLACE_ID=true
      - DB_NAME=user-management
      - DB_PORT=27017
      - DB_HOST=mongo
    volumes:
      - ./datastore:/datastore
    working_dir: /datastore
    links:
      - mongo
    depends_on:
      - mongo

  server:
    image: 'node:14.6.0'
    volumes:
      - ./server:/server
    working_dir: /server
    ports:
      - "3000:3000"
    environment:
      - PORT=3000
      - CLIENT_ORIGIN=http://localhost:8080
      - DB_CONNECTION_STRING=mongodb://mongo:27017/user-management
      - OVERNIGHT_JWT_SECRET="my super long random string for fake auth"
      - OVERNIGHT_JWT_EXP=24h
    command: npm start
    depends_on:
      - mongo

  client:
    image: node:16-alpine 
    container_name: user-management-client
    working_dir: /app
    volumes:
      - ./client:/app
    ports:
      - '8080:8080'
    command: sh -c "npm install --legacy-peer-deps && npm start"
    depends_on:
      - server
  
  nginx:
    image: nginx:alpine
    container_name: user-management-nginx
    ports:
      - '80:80'
    volumes:
      - ./nginx.conf:/etc/nginx/conf.d/default.conf:ro
      - ./logs/nginx:/var/log/nginx
    depends_on:
      - client
      - server
